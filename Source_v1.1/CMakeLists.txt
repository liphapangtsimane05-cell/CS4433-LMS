cmake_minimum_required(VERSION 3.16)
project(CS4433_Moderated_LMS VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(lms_core src/db.cpp src/security.cpp src/repository.cpp src/service.cpp)
target_include_directories(lms_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lms_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
if(MSVC)
 target_compile_options(lms_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(lms_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(lms src/main.cpp)
target_link_libraries(lms PRIVATE lms_core)
add_custom_command(TARGET lms POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:lms>/sql" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/sql/schema.sql" "$<TARGET_FILE_DIR:lms>/sql/schema.sql")
include(CTest)
if(BUILD_TESTING)
 add_executable(lms_tests tests/smoke.cpp)
 target_link_libraries(lms_tests PRIVATE lms_core)
 target_compile_definitions(lms_tests PRIVATE LMS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME LMS_Business_Workflows COMMAND lms_tests)
endif()

# Gate 3: instrument business logic and run framework-based unit tests.
option(LMS_COVERAGE "Measure GCC/Clang source-line coverage" OFF)
option(LMS_CATCH2_TESTS "Enable Catch2 unit tests" OFF)
if(LMS_COVERAGE)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "LMS_COVERAGE needs GCC or Clang")
 endif()
 target_compile_options(lms_core PRIVATE --coverage -O0 -g)
 target_link_options(lms_core INTERFACE --coverage)
endif()
if(BUILD_TESTING AND LMS_CATCH2_TESTS)
 include(FetchContent)
 FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.7.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(Catch2)
 add_executable(lms_unit_tests tests/unit_catch2.cpp)
 target_link_libraries(lms_unit_tests PRIVATE lms_core Catch2::Catch2WithMain)
 add_test(NAME LMS_Unit_Catch2 COMMAND lms_unit_tests)
endif()
